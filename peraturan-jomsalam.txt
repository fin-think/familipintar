=== FIRESTORE RULES: JomSalam ===
Tampal di dalam `match /databases/{database}/documents { ... }`, SELEPAS blok admin ditutup
(sama seperti blok buatapa_*). Kemudian Publish.

    // ----- JomSalam -----
    function slLink(u) { return get(/databases/$(database)/documents/mynasab_links/$(u)).data; }
    function slAdaLink(u) { return exists(/databases/$(database)/documents/mynasab_links/$(u)) && slLink(u).status == 'aktif'; }
    function slSekeluarga(a, b) {
      return (slAdaLink(a) && slLink(a).owner_uid == b)
          || (slAdaLink(b) && slLink(b).owner_uid == a)
          || (slAdaLink(a) && slAdaLink(b) && slLink(a).owner_uid == slLink(b).owner_uid);
    }

    // Indeks carian: {sha256(emel)} atau kod_XXXXXX -> { uid, nama }. Hanya get (tiada senarai).
    match /salam_cari/{id} {
      allow get: if request.auth != null;
      allow list: if false;
      allow create: if request.auth != null && request.resource.data.uid == request.auth.uid
                    && (id == hashing.sha256(request.auth.token.email.lower()).toHexString().lower() || id.matches('kod_[A-Z0-9]{6}'));
      allow update: if request.auth != null && resource.data.uid == request.auth.uid && request.resource.data.uid == request.auth.uid;
      allow delete: if request.auth != null && resource.data.uid == request.auth.uid;
    }
    match /salam_kod/{uid} {
      allow read, write: if request.auth != null && request.auth.uid == uid;
    }

    match /salam_chat/{cid} {
      allow read: if request.auth != null && (resource == null || request.auth.uid in resource.data.ahli);
      allow create: if request.auth != null
        && request.resource.data.ahli.size() == 2
        && request.resource.data.ahli[0] < request.resource.data.ahli[1]
        && cid == request.resource.data.ahli[0] + '_' + request.resource.data.ahli[1]
        && request.auth.uid in request.resource.data.ahli
        && request.resource.data.pemohon == request.auth.uid
        && ((request.resource.data.jenis == 'adam' && request.resource.data.status == 'menunggu')
            || (request.resource.data.jenis == 'keluarga' && request.resource.data.status == 'aktif'
                && slSekeluarga(request.resource.data.ahli[0], request.resource.data.ahli[1])));
      allow update: if request.auth != null && request.auth.uid in resource.data.ahli
        && request.resource.data.ahli == resource.data.ahli
        && request.resource.data.jenis == resource.data.jenis
        && request.resource.data.pemohon == resource.data.pemohon
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['mesej_akhir', 'masa_akhir', 'dari_akhir', 'belum', 'sembunyi', 'status', 'disekat_oleh'])
        && (
          (request.resource.data.status == resource.data.status
             && (resource.data.status == 'aktif' || !request.resource.data.diff(resource.data).affectedKeys().hasAny(['mesej_akhir', 'masa_akhir', 'dari_akhir'])))
          || (resource.data.status == 'menunggu' && request.resource.data.status in ['aktif', 'ditolak'] && request.auth.uid != resource.data.pemohon)
          || (request.resource.data.status == 'disekat' && request.resource.data.disekat_oleh == request.auth.uid)
          || (resource.data.status == 'disekat' && request.resource.data.status == 'aktif' && resource.data.disekat_oleh == request.auth.uid)
        );
      allow delete: if false;

      match /mesej/{mid} {
        function slChat() { return get(/databases/$(database)/documents/salam_chat/$(cid)).data; }
        allow read: if request.auth != null && request.auth.uid in slChat().ahli;
        allow create: if request.auth != null && request.auth.uid in slChat().ahli
          && slChat().status == 'aktif'
          && request.resource.data.dari == request.auth.uid
          && request.resource.data.keys().hasOnly(['dari', 'teks', 'at', 'luput_pada'])
          && request.resource.data.teks is string && request.resource.data.teks.size() > 0 && request.resource.data.teks.size() <= 2000
          && (slChat().jenis == 'keluarga' || request.resource.data.luput_pada is timestamp);
        allow update: if false;
        allow delete: if request.auth != null && resource.data.dari == request.auth.uid;
      }
    }

    match /salam_laporan/{id} {
      allow create: if request.auth != null && request.resource.data.pelapor == request.auth.uid;
      allow read, update, delete: if false;   // admin sahaja (blok admin)
    }

=== TTL (padam automatik mesej Kita Nasab Adam selepas 90 hari) ===
Google Cloud Console -> Firestore -> Time-to-live (TTL) -> Create policy
  Collection group: mesej
  Timestamp field:  luput_pada
Mesej Keluarga tiada medan luput_pada, jadi tidak dipadam.

Pengguna tanpa tarikh lahir akan diminta mengisi dob (untuk had 18 tahun Kita Nasab Adam).
Pastikan rule mynasab_users membenarkan pemilik mengemas kini medan dob.
