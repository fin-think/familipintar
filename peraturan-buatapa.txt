function ahliKumpulan(g) {
  return request.auth != null && (
    g == request.auth.uid ||
    (exists(/databases/$(database)/documents/mynasab_links/$(request.auth.uid)) &&
     get(/databases/$(database)/documents/mynasab_links/$(request.auth.uid)).data.owner_uid == g &&
     get(/databases/$(database)/documents/mynasab_links/$(request.auth.uid)).data.status == 'aktif') ||
    (g.matches('.*_kami') &&
     exists(/databases/$(database)/documents/buatapa_kumpulan/$(g)) &&
     request.auth.uid in get(/databases/$(database)/documents/buatapa_kumpulan/$(g)).data.ahli));
}

match /buatapa_kumpulan/{id} {
  allow read:   if request.auth != null && request.auth.uid in resource.data.ahli;
  allow create, update: if request.auth != null && id == request.auth.uid + '_kami'
                && request.resource.data.owner_uid == request.auth.uid
                && request.auth.uid in request.resource.data.ahli;
  allow delete: if false;
}

match /buatapa_posts/{id} {
  allow read:   if ahliKumpulan(resource.data.group_id);
  allow create: if request.resource.data.author_uid == request.auth.uid
                && ahliKumpulan(request.resource.data.group_id)
                && request.resource.data.teks.size() <= 2000;
  allow delete: if resource.data.author_uid == request.auth.uid
                || resource.data.group_id == request.auth.uid
                || resource.data.group_id == request.auth.uid + '_kami';
  allow update: if false;
}
match /buatapa_komen/{id} {
  allow read:   if ahliKumpulan(resource.data.group_id);
  allow create: if request.resource.data.author_uid == request.auth.uid
                && ahliKumpulan(request.resource.data.group_id)
                && request.resource.data.teks.size() <= 500;
  allow delete: if resource.data.author_uid == request.auth.uid
                || resource.data.group_id == request.auth.uid
                || resource.data.group_id == request.auth.uid + '_kami';
  allow update: if false;
}
match /buatapa_reaksi/{id} {
  allow read:   if ahliKumpulan(resource.data.group_id);
  allow create, update: if request.resource.data.uid == request.auth.uid
                && id == request.resource.data.post_id + '_' + request.auth.uid
                && ahliKumpulan(request.resource.data.group_id);
  allow delete: if resource.data.uid == request.auth.uid;
}

match /buatapa_profil/{uid} {
  allow read:   if request.auth != null;
  allow create, update, delete: if request.auth != null && request.auth.uid == uid;
}
